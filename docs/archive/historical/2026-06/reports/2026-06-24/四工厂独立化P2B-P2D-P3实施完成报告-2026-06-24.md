### Market Event Ingest Orchestrator 核心流程

```python
# packages/strategy-factory/src/strategy_factory/application/market_event_ingest/orchestrator.py

class MarketEventIngestOrchestrator:
    """Phase 2D: Bridge contract 和 normalization logic"""
    
    async def run_cycle(self, *, trigger: str, lookback_days: int) -> dict:
        # Phase 1: Fetch
        raw_events = await self._support.fetch_official_market_events(db, lookback_days=lookback_days)
        
        # Phase 2: Normalize & Bridge
        for raw_event in raw_events:
            # Normalization decision (Strategy Factory owns)
            decision = self._make_normalization_decision(raw_event)
            
            if decision.should_normalize:
                await self._support.save_normalized_event(db, normalized)
                
                # Bridge decision (Strategy Factory owns)
                bridge_payload = self._make_bridge_decision(normalized, raw_event)
                
                if bridge_payload.bridge_decision == "cluster":
                    await self._support.save_factory_event_cluster(db, cluster_data)
                elif bridge_payload.bridge_decision == "signal":
                    await self._support.save_factory_event_signal(db, signal_data)
```

```python
# packages/strategy-factory/src/strategy_factory/runtime/incubation.py

from ..application.incubation import IncubationOrchestrator

class IncubationRuntime:
    def __init__(self, support: Any):
        self._support = support
        self._orchestrator = IncubationOrchestrator(support) if support else None
    
    async def run_once(self, *, trigger: str, dry_run: bool) -> dict:
        if self._orchestrator is None:
            return {"success": False, "error": "incubation_support_missing"}
        return await self._orchestrator.run_cycle(trigger=trigger, dry_run=dry_run)
```
