| 入口 | 文件 | 用途 | 备注 |

| `StrategyFactoryScheduler` | `packages/strategy-factory/src/strategy_factory/application/factory_scheduler.py` | 主实现：被 AIASK Agent / MCP server / DesktopApp / BFF 加载 | 长期持有；持久化 EMA、运行结果、Top-N artifact 等 |

## 职责

调度器是策略工厂的入口，负责：
1. 按时间间隔循环触发工厂周期
2. 管理断路器（连续失败保护）
3. 维护 family 级别的 Gate 通过率反馈（EMA 平滑）
4. 协调所有 Gateway 实例的生命周期
5. 管理 TaskBoard 任务追踪

## 运行模式

| 模式 | 触发方式 | 间隔计算 |
|---|---|---|
| `continuous`（默认） | `_loop()` 无限循环 | 盘中 720s / 盘后 3600s / 周末 3600s |
| `daily` | `_loop()` 计算到下一个 run_time 的等待时间 | 每日一次 |

### 盘中/盘后判定

```python
@staticmethod
def _is_market_hours(now: datetime) -> bool:
    """A 股盘中时间：工作日 9:30-15:00"""
    if now.weekday() >= 5:  # 周六日
        return False
    t = now.time()
    return time(9, 30) <= t < time(15, 0)
```

### 等待时间计算

```python
def _compute_next_wait(self, now: datetime) -> float:
    # 首次运行使用启动延迟
    if self._cycle_count == 0:
        return float(AUTONOMY_STARTUP_DELAY_SEC)  # 默认 0

    if self.schedule_mode == "daily":
        # 计算到下一个 run_time 的秒数
        target = datetime.combine(now.date(), self.run_time, tzinfo=self._market_timezone)
        if target <= now:
            target += timedelta(days=1)
        return (target - now).total_seconds()

    # continuous 模式
    if now.weekday() >= 5:          # 周末
        return 3600.0
    if self._is_market_hours(now):   # 盘中
        return 720.0                 # 12 分钟
    return 3600.0                    # 盘后 1 小时
```

## `_loop()` 主循环（实际代码）

```python
async def _loop(self):
    while self._running:
        try:
            now = self._now()
            today_str = now.strftime("%Y-%m-%d")

            # 1. 日期变更 → 重置每日计数
            if self._daily_run_date != today_str:
                self._daily_run_date = today_str
                self._daily_run_count = 0

            # 2. 断路器打开 → 等待恢复
            if self._circuit_open_until is not None and now < self._circuit_open_until:
                sleep_sec = (self._circuit_open_until - now).total_seconds()
                await asyncio.sleep(min(sleep_sec + 1, 300))
                continue

            # 3. 达到每日上限 → 睡到午夜
            if self._daily_run_count >= self.max_daily_runs:
                tomorrow = datetime.combine(now.date() + timedelta(days=1), time(0, 0))
                sleep_sec = (tomorrow - now).total_seconds() + 1
                await asyncio.sleep(sleep_sec)
                continue

            # 4. 计算等待时间并睡眠
            wait = self._compute_next_wait(now)
            await asyncio.sleep(wait)

            # 5. 执行一轮
            if self._running:
                await self.run_once()
                self._daily_run_count += 1
                self._cycle_count += 1
                # 成功 → 重置断路器
                self._consecutive_failures = 0
                self._circuit_open_until = None

        except asyncio.CancelledError:
            break
        except Exception as exc:
            # 6. 失败处理 + 断路器
            self._consecutive_failures += 1
            if self._consecutive_failures >= self._max_consecutive_failures:
                self._circuit_open_until = self._now() + timedelta(
                    seconds=self._circuit_open_backoff_sec
                )
            await asyncio.sleep(FACTORY_ERROR_BACKOFF_SEC)
```

## `run_once()` 单次执行（实际代码）

```python
async def run_once(self, db=None, *, execution_mode=None, dispatch_id=None) -> dict:
    """执行一次完整的策略工厂流程。"""
    async with self._get_run_once_task_lock():
        # 防止并发重入：如果已有 run_once 在执行，等待其完成
        task = self._run_once_task
        if task is not None and not task.done():
            # 加入已有执行
            ...
        else:
            async def _execute_once() -> dict:
                resolved_db = self._load_db() if db is None else db
                previous_result = self.last_result

                # TaskBoard 任务追踪
                board_task = task_board.create_task(...)
                task_board.claim_task(...)

                # ★ 核心：执行工厂周期
                results, persistence_failures = await self._execute_factory_run_once_mode(
                    resolved_db,
                    previous_result=previous_result,
                    execution_mode=resolved_mode,
                )

                # 持久化运行结果
                self.last_run = self._now()
                self.last_result = results
                await self._persist_run_result(resolved_db, results, ...)

                # TaskBoard 完成
                task_board.complete_task(...)

                # P2-D：更新 family gate 反馈 EMA（α=0.3）
                family_counts = results["summary"]["incubation_budget_family_counts"]
                for family, count in family_counts.items():
                    prev_ema = self._family_gate_feedback[family]["ema_submit_count"]
                    new_ema = 0.3 * count + 0.7 * prev_ema
                    self._family_gate_feedback[family] = {"ema_submit_count": new_ema}
                # 衰减未出现的 family
                for family in self._family_gate_feedback:
                    if family not in family_counts:
                        self._family_gate_feedback[family]["ema_submit_count"] *= 0.7

                return results

            task = asyncio.create_task(_execute_once())
            self._run_once_task = task
```

## `_execute_factory_run_once_mode()` 执行模式

```python
async def _execute_factory_run_once_mode(self, resolved_db, *, previous_result, execution_mode):
    # 1. 执行主周期
    primary_result, persistence_failures = await self._execute_factory_cycle_once(
        resolved_db,
        execution_mode=resolved_mode,
        parity_role="primary",
        read_only=False,
    )

    # 2. 执行模拟盘日循环（PR-D1）
    paper_result = await run_paper_trading_cycle(resolved_db)
    primary_result["paper_trading_cycle"] = paper_result

    # 3. 如果是 SHADOW_READONLY 模式，额外执行一次只读影子运行
    if resolved_mode == FactoryExecutionMode.SHADOW_READONLY:
        shadow_result = await self._execute_factory_cycle_once(
            resolved_db, parity_role="shadow", read_only=True,
        )
        primary_result["parity_result"] = build_shadow_parity_result(primary_result, shadow_result)

    return primary_result, persistence_failures
```

```python
async def _execute_factory_cycle_once(self, resolved_db, *, execution_mode, parity_role, read_only):
    context = FactoryRunContext(
        db=resolved_db,
        factory_pkg=get_strategy_factory_package(),
        runtime_adapters=self._runtime_adapters,
        start=self._now(),
        trace_id=f"strategy_factory:{uuid4().hex[:12]}",
```

| 参数 | 环境变量 | 默认值 |

- `run_once()` 成功 → 重置 `_consecutive_failures = 0`

## 已识别的优化方向

> 以下为经过代码审计和行业最佳实践对比后识别的改进点，按优先级排列。

---

### 优化 1：中国节假日感知（🔴 高优先级）

**现状问题**：`_is_market_hours` 仅判断 `weekday() >= 5`（周六日），不识别中国法定节假日（春节、国庆等长假）和调休工作日。

- 春节/国庆等长假期间，调度器仍按"工作日盘后"间隔（3600s）运行，白白消耗资源
- 调休日（周六上班）被错误判定为周末，错过盘中高频运行窗口

**建议方案**：集成 `cn-stock-holidays` 或 `pandas_market_calendars` 库，或维护本地交易日历：

```python
def _is_trading_day(self, now: datetime) -> bool:
    """判断是否为 A 股交易日（排除节假日，包含调休日）。"""
    if self._trading_calendar is None:
        return now.weekday() < 5  # 降级为当前逻辑
    return self._trading_calendar.is_trading_day(now.date())

def _compute_next_wait(self, now: datetime) -> float:
    if self._cycle_count == 0:
        return float(AUTONOMY_STARTUP_DELAY_SEC)

    if self.schedule_mode == "daily":
        target = datetime.combine(now.date(), self.run_time, tzinfo=self._market_timezone)
        if target <= now:
            target += timedelta(days=1)
        # 跳过非交易日
        while not self._is_trading_day(target):
            target += timedelta(days=1)
        return (target - now).total_seconds()

    # continuous 模式
    if not self._is_trading_day(now):
        return float(self._non_trading_day_interval_sec)  # 建议 7200s
    if self._is_market_hours(now):
        return float(FACTORY_MARKET_HOURS_INTERVAL_SEC)
    return float(FACTORY_OFF_HOURS_INTERVAL_SEC)
```

**影响**：避免节假日无效运行，节省计算资源；调休日不再错过盘中窗口。

### 优化 2：断路器增加 Half-Open 状态（🔴 高优先级）

**现状问题**：断路器只有 CLOSED（正常）和 OPEN（阻断）两个状态。冷却期结束后直接恢复全量运行，如果底层问题未完全恢复，可能立即再次触发断路器。

**建议方案**：增加标准的 HALF-OPEN 状态，恢复后先执行一次轻量级探测：

```python
# 新增状态字段
self._circuit_state: str = "closed"  # closed / open / half_open

async def _loop(self):
    while self._running:
        try:
            # ... 现有日期/上限检查 ...

            # 断路器状态机
            if self._circuit_state == "open":
                if self._now() >= self._circuit_open_until:
                    self._circuit_state = "half_open"
                    logger.info("StrategyFactory: circuit breaker → HALF_OPEN, probing...")
                else:
                    await asyncio.sleep(min(remaining + 1, 300))
                    continue

            if self._circuit_state == "half_open":
                try:
                    await self.run_once()
                    # 探测成功 → 完全恢复
                    self._circuit_state = "closed"
                    self._consecutive_failures = 0
                    self._circuit_open_backoff_sec = 1800  # 重置退避时间
                except Exception:
                    # 探测失败 → 重新打开，指数退避
                    self._circuit_open_backoff_sec = min(
                        self._circuit_open_backoff_sec * 2, 7200
                    )
                    self._circuit_state = "open"
                    self._circuit_open_until = self._now() + timedelta(
                        seconds=self._circuit_open_backoff_sec
                    )
                continue

            # 正常 closed 状态执行
            wait = self._compute_next_wait(now)
            await asyncio.sleep(wait)
            if self._running:
                await self.run_once()
                # ...
```

**影响**：避免恢复后立即再次崩溃；指数退避防止频繁探测加重下游压力。

### 优化 3：EMA 反馈饥饿保护（🟡 中优先级）

**现状问题**：未出现的 family 每轮衰减 ×0.7，长期不被选中的 family 的 EMA 趋近于 0。`IncubationBudgeter` 中 `ema_submit < 0.3 → score -= 2.0`，形成正反馈死循环——越不被选中越不可能被选中。

**建议方案**：

```python
# 方案 A：设置 EMA 下限（推荐）
MIN_EMA_FLOOR = 0.15
for family in list(self._family_gate_feedback):
    if family not in family_counts:
        prev_ema = self._family_gate_feedback[family].get("ema_submit_count", 0.0)
        new_ema = max(MIN_EMA_FLOOR, prev_ema * 0.7)
        self._family_gate_feedback[family]["ema_submit_count"] = round(new_ema, 4)

# 方案 B：周期性探索重置（补充）
if self._cycle_count % 20 == 0:
    for family, data in self._family_gate_feedback.items():
        if data.get("ema_submit_count", 0) < 0.2:
            data["ema_submit_count"] = 0.5  # 给予重新探索的机会
```

**影响**：防止 family 被永久边缘化，保持策略多样性。

### 优化 4：EMA 反馈状态持久化（🟡 中优先级）

**现状问题**：`_family_gate_feedback` 是纯内存字典，进程重启后丢失所有历史反馈，需要多轮才能重建有效信号。

**建议方案**：

```python
# 在 _persist_run_result 中同时持久化反馈状态
async def _persist_run_result(self, db, results, ...):
    # ... 现有持久化逻辑 ...
    try:
        await db.save_scheduler_state({
            "family_gate_feedback": self._family_gate_feedback,
            "cycle_count": self._cycle_count,
            "consecutive_failures": self._consecutive_failures,
            "circuit_state": self._circuit_state,
        })
    except Exception as exc:
        logger.debug("scheduler state persistence failed: %s", exc)

# 启动时恢复
async def _restore_scheduler_state(self, db):
    try:
        state = await db.load_scheduler_state()
        if state:
            self._family_gate_feedback = state.get("family_gate_feedback", {})
            self._cycle_count = state.get("cycle_count", 0)
    except Exception:
        pass  # 降级为空状态启动
```

**影响**：重启后保留历史反馈信息，避免冷启动期间的盲目分配。

**现状问题**：调度器仅通过 `logger` 输出日志，没有结构化指标暴露，无法实时监控健康状态或设置告警。

### 优化 6：错误退避增加指数退避 + 抖动（🟡 中优先级）

**现状问题**：每次失败后固定等待 120s，不区分错误类型，无抖动机制。

**建议方案**：

```python
def _compute_error_backoff(self) -> float:
    """指数退避 + 抖动，避免惊群效应。"""
    base = FACTORY_ERROR_BACKOFF_SEC  # 120s
    # 指数退避：120s → 240s → 480s → 960s（上限 1800s）
    backoff = min(base * (2 ** (self._consecutive_failures - 1)), 1800)
    # 添加 ±20% 抖动
    jitter = backoff * random.uniform(-0.2, 0.2)
    return max(30.0, backoff + jitter)

# 在 _loop 异常处理中使用
except Exception as exc:
    self._consecutive_failures += 1
    # ...断路器逻辑...
    await asyncio.sleep(self._compute_error_backoff())
```

**影响**：瞬时错误快速恢复，持久性故障逐步退让；多实例部署时避免同时重试。

### 优化 7：优雅关闭改进（🟢 低优先级）

**现状问题**：`_shutdown_grace_sec = 0.25`（250ms），如果 `run_once` 正在执行（30-120s），几乎不可能等到完成，强制取消可能导致持久化不完整。

**建议方案**：

```python
async def shutdown(self):
    self._running = False
    task = self._task
    self._task = None
    if task is None:
        return

    if not task.done():
        # 第一阶段：等待当前周期自然完成（最多 120s）
        try:
            await asyncio.wait_for(asyncio.shield(task), timeout=120.0)
        except asyncio.TimeoutError:
            # 第二阶段：发送取消信号，等待清理（最多 5s）
            logger.warning("StrategyFactory: shutdown timeout, cancelling...")
            task.cancel()
            with suppress(asyncio.CancelledError):
                await asyncio.wait_for(task, timeout=5.0)
    else:
        with suppress(asyncio.CancelledError):
            await task
    logger.info("StrategyFactory stopped gracefully")
```

**影响**：避免关闭时数据不一致，确保最后一轮持久化完成。

### 优化 8：自适应调度间隔（🟢 低优先级）

**现状问题**：盘中固定 720s、盘后固定 3600s，不根据市场活跃度或工厂产出效率动态调整。

**建议方案**：

```python
def _compute_next_wait(self, now: datetime) -> float:
    # ... 现有逻辑 ...

    # 自适应调整：如果最近 N 轮产出率低，适当延长间隔
    if self._is_market_hours(now):
        base_interval = float(FACTORY_MARKET_HOURS_INTERVAL_SEC)
        # 最近 5 轮的平均提交数
        recent_submit_avg = self._get_recent_submit_average(window=5)
        if recent_submit_avg < 1.0:
            # 产出率低 → 延长 50%（最多 1080s）
            return min(base_interval * 1.5, 1080.0)
        elif recent_submit_avg > 5.0:
            # 产出率高 → 缩短 25%（最少 360s）
            return max(base_interval * 0.75, 360.0)
        return base_interval
```

**影响**：市场活跃时加快迭代，低效时节省资源。
